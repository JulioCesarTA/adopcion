export interface MlDetail {

  /** Probabilidad estimada de adopción sostenida, en [0, 1]. */
  probability: number;
  /** Fracción de árboles que coincide con la clase mayoritaria (confianza). */

  factors: {
    feature: string;
    label: string;
    value: number;
    contribution: number;
    direction: string;
  }[];
}
